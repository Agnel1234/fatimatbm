-- SECTION 4: INSERT FAMILY MEMBERS (3-7 members per family)
------------------------------------------------------------------------------------------------------

-- Create temp table for member names
CREATE TABLE #TempMemberNames (ID INT IDENTITY(1,1), Name NVARCHAR(50));

INSERT INTO #TempMemberNames VALUES ('John'), ('James'), ('Michael'), ('David'), ('Robert'), ('William'), ('Richard'), ('Charles'),
    ('Mary'), ('Anna'), ('Sarah'), ('Elizabeth'), ('Martha'), ('Ruth'), ('Rachel'), ('Rebecca'), ('Jennifer'), ('Patricia'),
    ('Linda'), ('Barbara'), ('Susan'), ('Jessica'), ('Lisa'), ('Nancy'), ('Karen'), ('Betty');

-- Insert family members
DECLARE @FamilyId INT;
DECLARE @MemberCount INT;
DECLARE @MemberIndex INT;

DECLARE family_cursor CURSOR FOR
    SELECT family_id FROM family WHERE family_code LIKE 'FAM%' ORDER BY family_id;

OPEN family_cursor;
FETCH NEXT FROM family_cursor INTO @FamilyId;

WHILE @@FETCH_STATUS = 0
BEGIN
    -- Generate member count (3-7 members per family)
    SET @MemberCount = 3 + (@FamilyId % 5);
    SET @MemberIndex = 1;

    WHILE @MemberIndex <= @MemberCount
    BEGIN
        DECLARE @MemberNameId INT = @MemberIndex;
        DECLARE @MemberName NVARCHAR(50) = (SELECT Name FROM #TempMemberNames WHERE ID = @MemberNameId);
        DECLARE @MemberRelationship NVARCHAR(30);
        DECLARE @MemberGender NVARCHAR(10);
        DECLARE @MemberDOB DATE;
        DECLARE @MemberEmail NVARCHAR(100);
        DECLARE @MemberPhone NVARCHAR(20);

        -- Assign relationships and gender
        IF @MemberIndex = 1
        BEGIN
            SET @MemberRelationship = 'Head';
            SET @MemberGender = 'Male';
        END
        ELSE IF @MemberIndex = 2
        BEGIN
            SET @MemberRelationship = 'Spouse';
            SET @MemberGender = 'Female';
        END
        ELSE IF @MemberIndex % 2 = 0
        BEGIN
            SET @MemberRelationship = 'Son';
            SET @MemberGender = 'Male';
        END
        ELSE
        BEGIN
            SET @MemberRelationship = 'Daughter';
            SET @MemberGender = 'Female';
        END

        -- Generate DOB (age 10-80)
        SET @MemberDOB = DATEADD(YEAR, -(10 + (@MemberIndex * 7) % 70), GETDATE());
        SET @MemberEmail = LOWER(@MemberName + '.' + CAST(@FamilyId AS NVARCHAR(10)) + '@fatimachurch.org');
        SET @MemberPhone = '98' + RIGHT('0000000' + CAST(75000000 + (@FamilyId * 54321) AS NVARCHAR(10)), 8);

        -- Insert family member
        INSERT INTO family_member (family_id, first_name, relationship, gender, dob, member_status, email, phone)
        VALUES (@FamilyId, @MemberName, @MemberRelationship, @MemberGender, @MemberDOB, 'Active', @MemberEmail, @MemberPhone);

        SET @MemberIndex = @MemberIndex + 1;
    END

    FETCH NEXT FROM family_cursor INTO @FamilyId;
END

CLOSE family_cursor;
DEALLOCATE family_cursor;

GO

    f.family_city,

-- Show subscription status summary by zone
SELECT
    a.anbiyam_name AS [Zone],
    COUNT(DISTINCT f.family_id) AS [Family Count],
    COUNT(DISTINCT fm.member_id) AS [Total Members],
    SUM(CASE WHEN fsy.jan_status = 'Paid' THEN 1 ELSE 0 END) AS [Jan Paid],
    SUM(CASE WHEN fsy.apr_status = 'Paid' THEN 1 ELSE 0 END) AS [Apr Paid],
    COUNT(DISTINCT cs.family_id) AS [Cemetery Subscriptions]
FROM family f
INNER JOIN anbiyam a ON f.anbiyam_id = a.anbiyam_id
LEFT JOIN family_member fm ON f.family_id = fm.family_id
LEFT JOIN family_subscription_yearly fsy ON f.family_id = fsy.family_id AND fsy.subscription_year = 2026
LEFT JOIN cemetery_subscription_yearly cs ON f.family_id = cs.family_id AND cs.subscription_year = 2026
WHERE f.family_code LIKE 'FAM%'
GROUP BY a.anbiyam_name
ORDER BY a.anbiyam_name;
